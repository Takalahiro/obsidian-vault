// 朝向：0=N(-y) 1=E(+x) 2=S(+y) 3=W(-x)

int heading = 0;

const int8_t DX[4] = {0, 1, 0, -1};

const int8_t DY[4] = {-1, 0, 1, 0};

  

const int GRID = 16;

uint8_t visited[GRID][2];

  

bool isVisited(int x, int y) {

if (x < 0 || y < 0 || x >= GRID || y >= GRID) return true;

return visited[y][x >> 3] & (1 << (x & 7));

}

void markVisited(int x, int y) {

if (x < 0 || y < 0 || x >= GRID || y >= GRID) return;

visited[y][x >> 3] |= (1 << (x & 7));

}

  

// T_CELL：正常直行穿过一格所需毫秒，必须用校准程序测出后填入！

const unsigned long T_CELL = 1480;

unsigned long straightStart = 0;

void beginStraightTiming() { straightStart = millis(); }

// ==================== 前墙停车补偿 ====================

// 前墙红外约提前一格报墙，确认前墙后先补走这一段再转向

const float FRONT_APPROACH_CELLS = 1.2f;

const int FRONT_APPROACH_PCT = 85;

const unsigned long FRONT_APPROACH_MS = (unsigned long)(FRONT_APPROACH_CELLS * T_CELL);

  

void advancePosition() {

unsigned long dt = millis() - straightStart;

int cells = (int)((dt + T_CELL / 2) / T_CELL);

if (cells < 1) cells = 1;

gx += DX[heading] * cells;

gy += DY[heading] * cells;

markVisited(gx, gy);

}

  

struct Frame { uint8_t px, py; uint8_t cameHeading; };

const int STACK_MAX = 40;

Frame stk[STACK_MAX];

int stkTop = -1;

bool exploreDone = false;

  

bool worthTrying(int d, bool openThisDir) {

if (d == (heading + 2) % 4) return false;

if (!openThisDir) return false;

return !isVisited(gx + DX[d], gy + DY[d]);

}

digitalWrite(LED_R, (s & 1) ? HIGH : LOW);

digitalWrite(LED_L, LOW);

digitalWrite(LED_R, (on && (s == 8 || s == 10)) ? HIGH : LOW);

digitalWrite(LED_M, (on && (s == 9 || s == 10)) ? HIGH : LOW);

}

digitalWrite(LED_L, on); digitalWrite(LED_M, on); digitalWrite(LED_R, on);

}

int irRead(int tx, int rx, unsigned long f) {

tone(tx, f); delayMicroseconds(700);

int v = digitalRead(rx);

noTone(tx); digitalWrite(tx, LOW); delayMicroseconds(250);

return v;

}

Reading readSensor(int tx, int rx) {

Reading res = {0, 0};

for (int b = 0; b < 5; b++) {

int hits = 0;

for (int s = 0; s < 3; s++) if (irRead(tx, rx, frequencies[b]) == LOW) hits++;

res.votes += hits;

if (hits >= 2) res.bands++;

}

return res;

}

void readAll(Reading &L, Reading &front, Reading &R) {

front = readSensor(FRONT_TX, FRONT_RX);

if (front.bands >= FRONT_WALL_THRESHOLD) stopRobot();

L = readSensor(LEFT_TX, LEFT_RX);

R = readSensor(RIGHT_TX, RIGHT_RX);

}

bool sideIsOpen(const Reading &r) { return r.bands <= SIDE_OPEN_THRESHOLD && r.votes <= SIDE_OPEN_MAX_VOTES; }

bool sideHasWall(const Reading &r) { return r.bands >= SIDE_WALL_THRESHOLD || r.votes >= SIDE_WALL_MIN_VOTES; }

  

// ==================== 串口 ====================

void printWait(const __FlashStringHelper *m) {

unsigned long now = millis();

if (now - lastWaitPrintTime < 500) return;

lastWaitPrintTime = now; Serial.println(m);

}

void printReadings(const Reading &L, const Reading &front, const Reading &R) {

unsigned long now = millis();

if (now - lastPrintTime < 200) return;

lastPrintTime = now;

Serial.print(F("L=")); Serial.print(L.bands); Serial.print('/'); Serial.print(L.votes);

Serial.print(F(" F=")); Serial.print(front.bands); Serial.print('/'); Serial.print(front.votes);

Serial.print(F(" R=")); Serial.print(R.bands); Serial.print('/'); Serial.println(R.votes);

}

void approachFrontWall() {

if (FRONT_APPROACH_MS == 0) return;

resetCorrection();

unsigned long t0 = millis();

while (millis() - t0 < FRONT_APPROACH_MS) {

Reading f = readSensor(FRONT_TX, FRONT_RX);

if (f.bands >= 5 && f.votes >= 14) break; // 贴墙硬保护

  

Reading L = readSensor(LEFT_TX, LEFT_RX);

Reading R = readSensor(RIGHT_TX, RIGHT_RX);

  

bool leftWall = sideHasWall(L);

bool rightWall = sideHasWall(R);

float rawError = 0.0f;

if (leftWall && rightWall) {

rawError = (float)(L.votes - ALIGN_LEFT_TARGET) - (float)(R.votes - ALIGN_RIGHT_TARGET);

} else if (leftWall) {

int e = L.votes - ALIGN_LEFT_TARGET; rawError = e > 0 ? (float)e : 0.0f;

} else if (rightWall) {

int e = R.votes - ALIGN_RIGHT_TARGET; rawError = e > 0 ? -(float)e : 0.0f;

}

filteredError = FILTER_OLD_WEIGHT * filteredError + (1.0f - FILTER_OLD_WEIGHT) * rawError;

  

int lOff = (LEFT_FORWARD - LEFT_STOP_PWM) * FRONT_APPROACH_PCT / 100;

int rOff = (RIGHT_STOP_PWM - RIGHT_FORWARD) * FRONT_APPROACH_PCT / 100;

int limit = (leftWall && rightWall) ? MAX_CORRECTION : SINGLE_WALL_MAX_CORRECTION;

int correction = 0;

if (filteredError > ERROR_DEADBAND || filteredError < -ERROR_DEADBAND)

correction = constrain((int)(filteredError * CORRECTION_GAIN), -limit, limit);

lastCorrection = correction;

motors(LEFT_STOP_PWM + lOff + correction, RIGHT_STOP_PWM - rOff + correction);

delay(5);

}

stopRobot();

resetCorrection();

delay(TURN_SETTLE_MS);

}

  
  

// ==================== 直行纠偏（原样保留） ====================

void driveStraight(const Reading &L, const Reading &front, const Reading &R) {

if (front.bands >= FRONT_WALL_THRESHOLD) { stopRobot(); resetCorrection(); return; }

unsigned long now = millis();

if (alignActive && now - alignStartedAt >= TURN_ALIGN_MS) alignActive = false;

  

bool leftWall = sideHasWall(L);

bool rightWall = sideHasWall(R);

int mask = (leftWall ? 2 : 0) | (rightWall ? 1 : 0);

if (mask != correctionWallMask) { resetCorrection(); correctionWallMask = mask; }

  

float rawError = 0.0f;

if (leftWall && rightWall) {

rawError = (float)(L.votes - ALIGN_LEFT_TARGET) - (float)(R.votes - ALIGN_RIGHT_TARGET);

} else if (leftWall) {

int e = L.votes - ALIGN_LEFT_TARGET; rawError = e > 0 ? (float)e : 0.0f;

} else if (rightWall) {

int e = R.votes - ALIGN_RIGHT_TARGET; rawError = e > 0 ? -(float)e : 0.0f;

}

filteredError = FILTER_OLD_WEIGHT * filteredError + (1.0f - FILTER_OLD_WEIGHT) * rawError;

if (!(leftWall && rightWall) && rawError == 0.0f) { filteredError = 0.0f; lastCorrection = 0; }

  

bool slow = now - startupTime < START_ALIGN_MS || alignActive || !leftWall || !rightWall ||

front.bands >= 1 || front.votes >= 3 || filteredError > 3.0f || filteredError < -3.0f;

int percent = slow ? CAUTION_DRIVE_PERCENT : 100;

int leftOffset = (LEFT_FORWARD - LEFT_STOP_PWM) * percent / 100;

int rightOffset = (RIGHT_STOP_PWM - RIGHT_FORWARD) * percent / 100;

  

int wanted = 0;

if (filteredError > ERROR_DEADBAND || filteredError < -ERROR_DEADBAND)

wanted = (int)(filteredError * CORRECTION_GAIN);

int limit = (leftWall && rightWall) ? MAX_CORRECTION : SINGLE_WALL_MAX_CORRECTION;

int motorLimit = max(0, min(leftOffset, rightOffset) - 5);

limit = min(limit, motorLimit);

wanted = constrain(wanted, -limit, limit);

int correction = constrain(wanted, lastCorrection - CORRECTION_STEP, lastCorrection + CORRECTION_STEP);

correction = constrain(correction, -limit, limit);

lastCorrection = correction;

  

motors(LEFT_STOP_PWM + leftOffset + correction, RIGHT_STOP_PWM - rightOffset + correction);

if (correction < 0) showScenario(5);

else if (correction > 0) showScenario(6);

else showScenario(1);

}

  

// ==================== 路口状态与故障 ====================

void clearJunctionTracking() { openAdvanceActive = false; openAdvanceStartedAt = 0; openAdvanceWallCount = 0; }

void stopWithFault(const __FlashStringHelper *m) { stopRobot(); resetCorrection(); motionFault = true; Serial.println(m); }

  

int sideState(int o, int w) {

if (o >= SIDE_CONFIRM_VOTES) return 1;

if (w >= SIDE_CONFIRM_VOTES) return 0;

return -1;

}

void checkJunction(int &leftState, int &rightState, bool &frontConfirmed) {

int lo=0, lw=0, ro=0, rw=0, fw=0;

stopRobot();

for (int i = 0; i < JUNCTION_SAMPLES; i++) {

Reading L, front, R; readAll(L, front, R);

if (sideIsOpen(L)) lo++; else if (sideHasWall(L)) lw++;

if (sideIsOpen(R)) ro++; else if (sideHasWall(R)) rw++;

if (front.bands >= FRONT_WALL_THRESHOLD) fw++;

showWaiting(); delay(10);

}

leftState = sideState(lo, lw);

rightState = sideState(ro, rw);

frontConfirmed = fw >= FRONT_CONFIRM_VOTES;

}

  

// ==================== 转弯 ====================

void markTurnFinished(bool armed) {

stopRobot(); resetCorrection(); clearJunctionTracking();

turnArmed = armed; clearVotes = 0; turnFinishedTime = millis();

alignActive = true; alignStartedAt = millis();

}

stopRobot(); resetCorrection(); showScenario(s);

lastNormalTurnLeft = true; delay(100);

spinLeft(); delay(d); stopRobot(); delay(TURN_SETTLE_MS); markTurnFinished(false);

}

void executeTurnRight(unsigned long d, int s) {

stopRobot(); resetCorrection(); showScenario(s);

lastNormalTurnLeft = false; delay(100);

spinRight(); delay(d); stopRobot(); delay(TURN_SETTLE_MS); markTurnFinished(false);

}

void executeScenario7() { showScenarioAndHold(7); spinRightFor(TIME_SMALL); forwardFor(TIME_AFTER_TURN); lastNormalTurnLeft = false; markTurnFinished(false); }

void executeScenario8() { showScenarioAndHold(8); spinLeftFor(TIME_SMALL); forwardFor(TIME_AFTER_TURN); lastNormalTurnLeft = true; markTurnFinished(false); }

  

// ==================== Case 9/10 ====================

void runEscapeMotor() {

if (escapeScenario == 9) {

if (escapeTurningLeft) motors(UTURN_LEFT_L, UTURN_LEFT_R); else motors(UTURN_RIGHT_L, UTURN_RIGHT_R);

} else {

if (escapeTurningLeft) spinLeft(); else spinRight();

}

}

void beginEscape(int s, bool turnLeft) {

stopRobot(); resetCorrection(); clearJunctionTracking(); alignActive = false;

delay(ACTION_PAUSE_MS); backupFor(ESCAPE_BACKUP_MS);

escapeScenario = s; escapeTurningLeft = turnLeft; escapeConfirming = false; escapeFrontClearCount = 0;

showScenario(s); Serial.print(F("ESCAPE: ")); Serial.println(s);

runEscapeMotor(); escapeStartedAt = millis();

}

void finishEscape() {

stopRobot(); escapeScenario = 0; escapeConfirming = false; escapeFrontClearCount = 0;

markTurnFinished(true);

beginStraightTiming(); // ← 加这行，掉头完成后才开始计直行时间

Serial.println(F("ESCAPE: corrected drive next frame"));

}

  

void updateEscape(const Reading &front) {

unsigned long now = millis();

showScenario(escapeScenario);

if (now - escapeStartedAt >= ESCAPE_TIMEOUT_MS) { stopWithFault(F("ESCAPE_TIMEOUT")); return; }

if (escapeScenario == 9 && !escapeConfirming) {

unsigned long dur = escapeTurningLeft ? UTURN_LEFT_MS : UTURN_RIGHT_MS;

if (now - escapeStartedAt < dur) return;

stopRobot(); escapeConfirming = true; escapeFrontClearCount = 0; escapeStoppedAt = millis();

return;

}

bool frontClear = front.bands == 0 && front.votes <= 1;

if (escapeConfirming) {

stopRobot();

if (now - escapeStoppedAt < ESCAPE_SETTLE_MS) return;

if (frontClear) {

escapeFrontClearCount++;

if (escapeFrontClearCount >= ESCAPE_CONFIRM_COUNT) finishEscape();

return;

}

escapeFrontClearCount = 0;

if (escapeScenario == 9) return;

escapeConfirming = false; runEscapeMotor(); return;

}

if (frontClear) { stopRobot(); escapeConfirming = true; escapeFrontClearCount = 0; escapeStoppedAt = millis(); return; }

runEscapeMotor();

}

  

// ==================== 初始化 ====================

void setup() {

Serial.begin(9600);

pinMode(LEFT_TX, OUTPUT); pinMode(FRONT_TX, OUTPUT); pinMode(RIGHT_TX, OUTPUT);

digitalWrite(LEFT_TX, LOW); digitalWrite(FRONT_TX, LOW); digitalWrite(RIGHT_TX, LOW);

pinMode(LEFT_RX, INPUT); pinMode(FRONT_RX, INPUT); pinMode(RIGHT_RX, INPUT);

pinMode(LED_L, OUTPUT); pinMode(LED_M, OUTPUT); pinMode(LED_R, OUTPUT);

ledsOff();

leftServo.attach(LEFT_SERVO); rightServo.attach(RIGHT_SERVO);

stopRobot(); delay(1000);

clearJunctionTracking();

startupTime = millis();

  

// 记忆层初始化

memset(visited, 0, sizeof(visited));

gx = 8; gy = 8; heading = 0;

markVisited(gx, gy);

stkTop = -1;

exploreDone = false;

beginStraightTiming();

  

Serial.println(F("READY: DFS explore, visited-memory"));

}

  

// ==================== 主循环 ====================

void loop() {

// 探索完成 → 永久停车

if (exploreDone) { stopRobot(); ledDone(); delay(20); return; }

  

// 故障锁定

if (motionFault) { stopRobot(); showWaiting(); printWait(F("FAULT")); delay(10); return; }

  

// Case 9 定时掉头

if (escapeScenario == 9 && !escapeConfirming) {

Reading unused = {0, 0}; updateEscape(unused); delay(1); return;

}

// Case 9/10 确认或搜索

if (escapeScenario == 9 || escapeScenario == 10) {

if (escapeConfirming && millis() - escapeStoppedAt < ESCAPE_SETTLE_MS) {

stopRobot(); showScenario(escapeScenario); delay(5); return;

}

Reading front = readSensor(FRONT_TX, FRONT_RX);

updateEscape(front); delay(5); return;

}

  

// 开放直行超时 → 停轮采样

if (openAdvanceActive && millis() - openAdvanceStartedAt >= OPEN_ADVANCE_TIMEOUT_MS) stopRobot();

  

Reading L, front, R;

readAll(L, front, R);

  

// ---------- 前墙最高优先级 ----------

if (front.bands >= FRONT_WALL_THRESHOLD) {

stopRobot(); resetCorrection(); clearJunctionTracking(); clearVotes = 0;

showWaiting(); delay(70);

  

int leftState = -1, rightState = -1; bool frontConfirmed = false;

checkJunction(leftState, rightState, frontConfirmed);

if (!frontConfirmed) { printWait(F("WAIT: FRONT_NOT_CONFIRMED")); return; }

  

// 掉头后立刻检测到前墙：跳过补走，直接决策

if (millis() - straightStart >= T_CELL / 2) {

approachFrontWall();

}

advancePosition();

  
  
  

// ====== 访问记忆 + Trémaux 岔口决策 ======

advancePosition();

  

int dL = (heading + 3) % 4;

int dF = heading;

int dR = (heading + 1) % 4;

bool openL = (leftState == 1);

bool openF = (!frontConfirmed); // 此处 frontConfirmed 必为 true，openF=false

bool openR = (rightState == 1);

  

bool okL = worthTrying(dL, openL);

bool okF = worthTrying(dF, openF);

bool okR = worthTrying(dR, openR);

  

// 转向锁未解除又撞确认前墙：沿用原 Case10

if (!turnArmed && !okL && !okF && !okR) {

if (leftState == 1 || rightState == 1) {

bool tl = (leftState == 1 && rightState == 1) ? lastNormalTurnLeft : (leftState == 1);

beginEscape(10, tl);

} else { showWaiting(); printWait(F("WAIT: TURN_LOCK")); }

return;

}

  

if (okL || okF || okR) {

if (stkTop < STACK_MAX - 1) {

stkTop++; stk[stkTop].px = gx; stk[stkTop].py = gy; stk[stkTop].cameHeading = heading;

}

ledExploring();

if (okL) { heading = dL; executeTurnLeft(LEFT_TURN_MS, 3); }

else if (okF) { markTurnFinished(true); }

else { heading = dR; executeTurnRight(RIGHT_TURN_MS, 2); }

beginStraightTiming();

return;

}

  

// 无可走 → 弹栈掉头回退；栈空 → 探索完成

if (stkTop < 0) {

exploreDone = true; stopRobot(); resetCorrection(); clearJunctionTracking();

ledDone(); Serial.println(F("EXPLORE DONE"));

return;

}

stkTop--;

ledBacktrack();

heading = (heading + 2) % 4;

beginEscape(9, true);

beginStraightTiming();

return;

}

  

// ---------- 转弯后重新解锁 ----------

if (!turnArmed) {

if (clearVotes < 3) clearVotes++;

if (clearVotes >= 3 && millis() - turnFinishedTime >= POST_TURN_REARM_MS) turnArmed = true;

}

  

// ---------- 侧面开口检测（捕获直行中的岔口） ----------

// 进入条件：已解锁、未在开放直行、某侧由"墙"变为"开放"

static bool prevLeftWall = false, prevRightWall = false;

bool lWall = sideHasWall(L), rWall = sideHasWall(R);

bool leftOpened = prevLeftWall && !lWall;

bool rightOpened = prevRightWall && !rWall;

  

if (turnArmed && !openAdvanceActive && (leftOpened || rightOpened)) {

stopRobot();

int leftState = -1, rightState = -1; bool frontConfirmed = false;

checkJunction(leftState, rightState, frontConfirmed);

  

advancePosition();

int dL = (heading + 3) % 4;

int dF = heading;

int dR = (heading + 1) % 4;

bool okL = worthTrying(dL, leftState == 1);

bool okF = worthTrying(dF, !frontConfirmed);

bool okR = worthTrying(dR, rightState == 1);

  

if (okL || okF || okR) {

if (stkTop < STACK_MAX - 1) {

stkTop++; stk[stkTop].px = gx; stk[stkTop].py = gy; stk[stkTop].cameHeading = heading;

}

ledExploring();

if (okL) { heading = dL; executeTurnLeft(LEFT_TURN_MS, 3); }

else if (okF) { markTurnFinished(true); }

else { heading = dR; executeTurnRight(RIGHT_TURN_MS, 2); }

beginStraightTiming();

prevLeftWall = false; prevRightWall = false;

return;

}

// 侧开口但无新可走格：忽略，继续直行（prevWall 已刷新）

prevLeftWall = sideHasWall(L); prevRightWall = sideHasWall(R);

// 落到下方 driveStraight

} else {

prevLeftWall = lWall; prevRightWall = rWall;

}

  

// ---------- 开放路口优先直行 ----------

bool bothOpen = sideIsOpen(L) && sideIsOpen(R);

bool bothWalls = sideHasWall(L) && sideHasWall(R);

  

if (!openAdvanceActive && bothOpen) {

openAdvanceActive = true; openAdvanceStartedAt = millis(); openAdvanceWallCount = 0;

resetCorrection(); Serial.println(F("OPEN: slow straight"));

}

if (openAdvanceActive) {

if (bothWalls) { if (openAdvanceWallCount < OPEN_EXIT_CONFIRM_COUNT) openAdvanceWallCount++; }

else openAdvanceWallCount = 0;

  

if (openAdvanceWallCount >= OPEN_EXIT_CONFIRM_COUNT) {

clearJunctionTracking(); resetCorrection();

alignActive = true; alignStartedAt = millis();

Serial.println(F("OPEN: corridor recovered"));

} else {

if (millis() - openAdvanceStartedAt >= OPEN_ADVANCE_TIMEOUT_MS) {

stopRobot(); clearJunctionTracking(); // 软停，不锁死

return;

}

motors(JUNCTION_FORWARD_L, JUNCTION_FORWARD_R);

showScenario(1); printReadings(L, front, R); delay(5);

return;

}

}

  

// ---------- 正常通道行驶（纠偏，原样保留） ----------

driveStraight(L, front, R);

printReadings(L, front, R);

delay(10);

}
