```cpp
// ★改1: 前墙补偿——双向纠偏 + 限速 + 防反转 + 入弯前冻结纠偏
void approachFrontWall() {
  if (FRONT_APPROACH_MS == 0) return;
  resetCorrection();
  unsigned long t0 = millis();
  while (millis() - t0 < FRONT_APPROACH_MS) {
    Reading f = readSensor(FRONT_TX, FRONT_RX);
    if (f.bands >= 5 && f.votes >= 14) break; // 贴墙硬保护

    Reading L = readSensor(LEFT_TX, LEFT_RX);
    Reading R = readSensor(RIGHT_TX, RIGHT_RX);

    bool leftWall = sideHasWall(L);
    bool rightWall = sideHasWall(R);

    // ★改1: 双向纠偏，以“有墙那一侧”为基准，太近推开、太远拉回（不再往空地飘）
    float rawError = 0.0f;
    if (leftWall && rightWall) {
      rawError = (float)(L.votes - ALIGN_LEFT_TARGET) - (float)(R.votes - ALIGN_RIGHT_TARGET);
    } else if (leftWall) {
      rawError =  (float)(L.votes - ALIGN_LEFT_TARGET);
    } else if (rightWall) {
      rawError = -(float)(R.votes - ALIGN_RIGHT_TARGET);
    }
    filteredError = FILTER_OLD_WEIGHT * filteredError + (1.0f - FILTER_OLD_WEIGHT) * rawError;
    if (!(leftWall && rightWall) && rawError == 0.0f) { filteredError = 0.0f; lastCorrection = 0; }

    int lOff = (LEFT_FORWARD - LEFT_STOP_PWM) * FRONT_APPROACH_PCT / 100;
    int rOff = (RIGHT_STOP_PWM - RIGHT_FORWARD) * FRONT_APPROACH_PCT / 100;

    int limit = (leftWall && rightWall) ? MAX_CORRECTION : SINGLE_WALL_MAX_CORRECTION;
    int motorLimit = max(0, min(lOff, rOff) - 5);   // ★改1: 防把轮子纠到反转
    limit = min(limit, motorLimit);

    int wanted = 0;
    if (filteredError > ERROR_DEADBAND || filteredError < -ERROR_DEADBAND)
      wanted = (int)(filteredError * CORRECTION_GAIN);
    wanted = constrain(wanted, -limit, limit);

    // ★改1: 限速，一格一格爬，消除进补偿瞬间的突跳
    int correction = constrain(wanted, lastCorrection - CORRECTION_STEP, lastCorrection + CORRECTION_STEP);
    correction = constrain(correction, -limit, limit);
    lastCorrection = correction;

    // ★改5: 入弯前最后 ~0.3 格冻结纠偏，纯直行，避免转弯偏航累积
    bool nearTurn = (millis() - t0) > (FRONT_APPROACH_MS > (T_CELL * 3 / 10)
                                       ? FRONT_APPROACH_MS - (T_CELL * 3 / 10) : 0);
    if (nearTurn) { motors(LEFT_STOP_PWM + lOff, RIGHT_STOP_PWM - rOff); lastCorrection = 0; }
    else          { motors(LEFT_STOP_PWM + lOff + correction, RIGHT_STOP_PWM - rOff + correction); }
    delay(5);
  }
  stopRobot();
  resetCorrection();
  delay(TURN_SETTLE_MS);
}

// ==================== 直行纠偏 ====================
void driveStraight(const Reading &L, const Reading &front, const Reading &R) {
  if (front.bands >= FRONT_WALL_THRESHOLD) { stopRobot(); resetCorrection(); return; }
  unsigned long now = millis();
  if (alignActive && now - alignStartedAt >= TURN_ALIGN_MS) alignActive = false;

  bool leftWall = sideHasWall(L);
  bool rightWall = sideHasWall(R);
  int mask = (leftWall ? 2 : 0) | (rightWall ? 1 : 0);
  if (mask != correctionWallMask) { resetCorrection(); correctionWallMask = mask; }

  float rawError = 0.0f;
  if (leftWall && rightWall) {
    rawError = (float)(L.votes - ALIGN_LEFT_TARGET) - (float)(R.votes - ALIGN_RIGHT_TARGET);
  } else if (leftWall) {
    // ★改2: 单墙改双向，离墙太远也拉回（不再单向漂）
    rawError =  (float)(L.votes - ALIGN_LEFT_TARGET);
  } else if (rightWall) {
    rawError = -(float)(R.votes - ALIGN_RIGHT_TARGET);
  }
  filteredError = FILTER_OLD_WEIGHT * filteredError + (1.0f - FILTER_OLD_WEIGHT) * rawError;
  if (!(leftWall && rightWall) && rawError == 0.0f) { filteredError = 0.0f; lastCorrection = 0; }

  bool slow = now - startupTime < START_ALIGN_MS || alignActive || !leftWall || !rightWall ||
              front.bands >= 1 || front.votes >= 3 || filteredError > 3.0f || filteredError < -3.0f;
  int percent = slow ? CAUTION_DRIVE_PERCENT : 100;
  int leftOffset = (LEFT_FORWARD - LEFT_STOP_PWM) * percent / 100;
  int rightOffset = (RIGHT_STOP_PWM - RIGHT_FORWARD) * percent / 100;

  int wanted = 0;
  if (filteredError > ERROR_DEADBAND || filteredError < -ERROR_DEADBAND)
    wanted = (int)(filteredError * CORRECTION_GAIN);
  int limit = (leftWall && rightWall) ? MAX_CORRECTION : SINGLE_WALL_MAX_CORRECTION;
  int motorLimit = max(0, min(leftOffset, rightOffset) - 5);
  limit = min(limit, motorLimit);
  wanted = constrain(wanted, -limit, limit);
  int correction = constrain(wanted, lastCorrection - CORRECTION_STEP, lastCorrection + CORRECTION_STEP);
  correction = constrain(correction, -limit, limit);
  lastCorrection = correction;

  motors(LEFT_STOP_PWM + leftOffset + correction, RIGHT_STOP_PWM - rightOffset + correction);
  if (correction < 0) showScenario(5);
  else if (correction > 0) showScenario(6);
  else showScenario(1);
}

    if (escapeTurningLeft) motors(UTURN_LEFT_L, UTURN_LEFT_R); else motors(UTURN_RIGHT_L, UTURN_RIGHT_R);

void beginEscape(int s, bool turnLeft) {
  stopRobot(); resetCorrection(); clearJunctionTracking(); alignActive = false;
  delay(ACTION_PAUSE_MS); backupFor(ESCAPE_BACKUP_MS);
  escapeScenario = s; escapeTurningLeft = turnLeft; escapeConfirming = false; escapeFrontClearCount = 0;

void finishEscape() {
  stopRobot(); escapeScenario = 0; escapeConfirming = false; escapeFrontClearCount = 0;
  markTurnFinished(true);
  beginStraightTiming();
  Serial.println(F("ESCAPE: corrected drive next frame"));

void updateEscape(const Reading &front) {
  unsigned long now = millis();
  showScenario(escapeScenario);
  if (now - escapeStartedAt >= ESCAPE_TIMEOUT_MS) { stopWithFault(F("ESCAPE_TIMEOUT")); return; }
  if (escapeScenario == 9 && !escapeConfirming) {
    unsigned long dur = escapeTurningLeft ? UTURN_LEFT_MS : UTURN_RIGHT_MS;
    if (now - escapeStartedAt < dur) return;
    stopRobot(); escapeConfirming = true; escapeFrontClearCount = 0; escapeStoppedAt = millis();
    return;
  }
  bool frontClear = front.bands == 0 && front.votes <= 1;
  if (escapeConfirming) {
    stopRobot();
    if (now - escapeStoppedAt < ESCAPE_SETTLE_MS) return;
    if (frontClear) {
      escapeFrontClearCount++;
      if (escapeFrontClearCount >= ESCAPE_CONFIRM_COUNT) finishEscape();
      return;
    }
    escapeFrontClearCount = 0;
    if (escapeScenario == 9) return;
    escapeConfirming = false; runEscapeMotor(); return;
  }
  if (frontClear) { stopRobot(); escapeConfirming = true; escapeFrontClearCount = 0; escapeStoppedAt = millis(); return; }
  runEscapeMotor();
}

void loop() {
  if (exploreDone) { stopRobot(); ledDone(); delay(20); return; }
  if (motionFault) { stopRobot(); showWaiting(); printWait(F("FAULT")); delay(10); return; }

  // Case 9 定时掉头
  if (escapeScenario == 9 && !escapeConfirming) {
    Reading unused = {0, 0}; updateEscape(unused); delay(1); return;
  }
  // Case 9/10 确认或搜索
  if (escapeScenario == 9 || escapeScenario == 10) {
    if (escapeConfirming && millis() - escapeStoppedAt < ESCAPE_SETTLE_MS) {
      stopRobot(); showScenario(escapeScenario); delay(5); return;
    }
    Reading front = readSensor(FRONT_TX, FRONT_RX);
    updateEscape(front); delay(5); return;
  }

  // 开放直行超时 → 停轮采样
  if (openAdvanceActive && millis() - openAdvanceStartedAt >= OPEN_ADVANCE_TIMEOUT_MS) stopRobot();

  Reading L, front, R;
  readAll(L, front, R);

  // ---------- 前墙最高优先级 ----------
  if (front.bands >= FRONT_WALL_THRESHOLD) {

    // ★改4(#1): U形弯第二弯防提前转——刚转完且前墙还不近时，先往前走过短边
    //   加 front.votes<10 闸：避免掉头后面对死墙时还往前冲
    if (millis() - turnFinishedTime < (T_CELL * 6 / 10) && front.votes < 10) {
      motors(JUNCTION_FORWARD_L, JUNCTION_FORWARD_R);
      delay(10);
      return;
    }

    stopRobot(); resetCorrection(); clearJunctionTracking(); clearVotes = 0;
    showWaiting(); delay(70);

    int leftState = -1, rightState = -1; bool frontConfirmed = false;
    checkJunction(leftState, rightState, frontConfirmed);
    if (!frontConfirmed) { printWait(F("WAIT: FRONT_NOT_CONFIRMED")); return; }

    // 掉头后立刻检测到前墙：跳过补走，直接决策
    if (millis() - straightStart >= T_CELL / 2) {
      approachFrontWall();
    }
    // ★改3(#4): 位置推算只调用一次（已删除重复的 advancePosition()）
    advancePosition();

    // ====== 访问记忆 + Trémaux 岔口决策 ======
    int dL = (heading + 3) % 4;
    int dF = heading;
    int dR = (heading + 1) % 4;
    bool openL = (leftState == 1);
    bool openF = (!frontConfirmed); // frontConfirmed 必为 true，openF=false
    bool openR = (rightState == 1);

    bool okL = worthTrying(dL, openL);
    bool okF = worthTrying(dF, openF);
    bool okR = worthTrying(dR, openR);

    // 转向锁未解除又撞确认前墙
    if (!turnArmed && !okL && !okF && !okR) {
      if (leftState == 1 || rightState == 1) {
        bool tl = (leftState == 1 && rightState == 1) ? lastNormalTurnLeft : (leftState == 1);
        beginEscape(10, tl);
      } else { showWaiting(); printWait(F("WAIT: TURN_LOCK")); }
      return;
    }

    if (okL || okF || okR) {
      if (stkTop < STACK_MAX - 1) {
        stkTop++; stk[stkTop].px = gx; stk[stkTop].py = gy; stk[stkTop].cameHeading = heading;
      }
      ledExploring();
      if (okL) { heading = dL; executeTurnLeft(LEFT_TURN_MS, 3); }
      else if (okF) { markTurnFinished(true); }
      else { heading = dR; executeTurnRight(RIGHT_TURN_MS, 2); }
      beginStraightTiming();
      return;
    }

    // 无可走 → 弹栈掉头回退；栈空 → 探索完成
    if (stkTop < 0) {
      exploreDone = true; stopRobot(); resetCorrection(); clearJunctionTracking();
      ledDone(); Serial.println(F("EXPLORE DONE"));
      return;
    }
    stkTop--;
    ledBacktrack();
    heading = (heading + 2) % 4;
    beginEscape(9, true);
    beginStraightTiming();
    return;
  }

  // ---------- 转弯后重新解锁 ----------
  if (!turnArmed) {
    if (clearVotes < 3) clearVotes++;
    if (clearVotes >= 3 && millis() - turnFinishedTime >= POST_TURN_REARM_MS) turnArmed = true;
  }

  // ---------- 侧面开口检测 ----------
  // ★改3: prevLeftWall/prevRightWall 已改为全局（在 markTurnFinished 里按真实读数重置）
  bool lWall = sideHasWall(L), rWall = sideHasWall(R);
  bool leftOpened = prevLeftWall && !lWall;
  bool rightOpened = prevRightWall && !rWall;

  if (turnArmed && !openAdvanceActive && (leftOpened || rightOpened)) {
    stopRobot();
    int leftState = -1, rightState = -1; bool frontConfirmed = false;
    checkJunction(leftState, rightState, frontConfirmed);

    advancePosition();
    int dL = (heading + 3) % 4;
    int dF = heading;
    int dR = (heading + 1) % 4;
    bool okL = worthTrying(dL, leftState == 1);
    bool okF = worthTrying(dF, !frontConfirmed);
    bool okR = worthTrying(dR, rightState == 1);

    if (okL || okF || okR) {
      if (stkTop < STACK_MAX - 1) {
        stkTop++; stk[stkTop].px = gx; stk[stkTop].py = gy; stk[stkTop].cameHeading = heading;
      }
      ledExploring();
      if (okL) { heading = dL; executeTurnLeft(LEFT_TURN_MS, 3); }
      else if (okF) { markTurnFinished(true); }
      else { heading = dR; executeTurnRight(RIGHT_TURN_MS, 2); }
      beginStraightTiming();
      prevLeftWall = false; prevRightWall = false;
      return;
    }
    // 侧开口但无新可走格：忽略，继续直行
    prevLeftWall = sideHasWall(L); prevRightWall = sideHasWall(R);
  } else {
    prevLeftWall = lWall; prevRightWall = rWall;
  }

  // ---------- 开放路口优先直行 ----------
  bool bothOpen = sideIsOpen(L) && sideIsOpen(R);
  bool bothWalls = sideHasWall(L) && sideHasWall(R);

  if (!openAdvanceActive && bothOpen) {
    openAdvanceActive = true; openAdvanceStartedAt = millis(); openAdvanceWallCount = 0;
    resetCorrection(); Serial.println(F("OPEN: slow straight"));
  }
  if (openAdvanceActive) {
    if (bothWalls) { if (openAdvanceWallCount < OPEN_EXIT_CONFIRM_COUNT) openAdvanceWallCount++; }
    else openAdvanceWallCount = 0;

    if (openAdvanceWallCount >= OPEN_EXIT_CONFIRM_COUNT) {
      clearJunctionTracking(); resetCorrection();
      alignActive = true; alignStartedAt = millis();
      Serial.println(F("OPEN: corridor recovered"));
    } else {
      if (millis() - openAdvanceStartedAt >= OPEN_ADVANCE_TIMEOUT_MS) {
        stopRobot(); clearJunctionTracking();
        return;
      }
      motors(JUNCTION_FORWARD_L, JUNCTION_FORWARD_R);
      showScenario(1); printReadings(L, front, R); delay(5);
      return;
    }
  }

  // ---------- 正常通道行驶 ----------
  driveStraight(L, front, R);
  printReadings(L, front, R);
  delay(10);
}
```

## ⚠️ 上机必须做的两件标定（我替不了）

1. **`ALIGN_LEFT_TARGET / ALIGN_RIGHT_TARGET = 7`**：现在改成双向纠偏后，这个数就是"居中目标"。串口看 `L=x/votes`，把车摆居中，读数是几就填几。**填错车会稳定偏一边。**

## 二代
**我会保留“连续直行、沿墙纠偏、路口左→前→右探索、短死路回退”的逻辑和你复测的所有标定值。不会再改成每格停车。**

但我不能再给你假的“百分百”保证：**没有实车验证，不能保证两张图必过。** 下面这版修正确定的程序错误；剩下的关键限制是，位置仍靠时间估算。第二张图也没标起终点，所以终点仍按你指定的“唯一长死路”识别。

这版具体改了：

- 前墙提前报墙时，进入**带纠偏的接近阶段**，不立即决定转向。
- **补走完成后重新采样**，不再使用补走前的左右墙判断。
- 侧开口需要连续确认，再补走到转弯位置。
- 补偿全程保留单墙／双墙纠偏，取消末段关闭纠偏。
- 删除“转完必须强行前进 0.6 格”的逻辑；前墙过近始终停车。
- 回溯实际使用节点的返回方向；遇到已经访问的节点会原路返回，避免把环路当新路无限走。
- 计时只累计电机前进的时间，停车采样和转弯不计入。
- 长死路必须**三向确认有墙、连续双侧有墙的行驶时间够长**，才停车。

```cpp
// ==================== 连续行驶纠偏 ====================
void driveCorrected(const Reading &L, const Reading &R,
                    int percent) {
  bool lw = hasWall(L);
  bool rw = hasWall(R);
  int mask = (lw ? 2 : 0) | (rw ? 1 : 0);

  if (mask != correctionWallMask) {
    resetCorrection();
    correctionWallMask = mask;
  }

  float error = 0;
  if (lw && rw) {
    error = (float)(L.votes - ALIGN_LEFT_TARGET)
          - (float)(R.votes - ALIGN_RIGHT_TARGET);
  } else if (lw) {
    error = (float)(L.votes - ALIGN_LEFT_TARGET);
  } else if (rw) {
    error = -(float)(R.votes - ALIGN_RIGHT_TARGET);
  }

  filteredError =
      FILTER_OLD_WEIGHT * filteredError +
      (1.0f - FILTER_OLD_WEIGHT) * error;

  if (!lw && !rw) {
    filteredError = 0;
    lastCorrection = 0;
  }

  int lOff = (LEFT_FORWARD - LEFT_STOP_PWM) * percent / 100;
  int rOff = (RIGHT_STOP_PWM - RIGHT_FORWARD) * percent / 100;

  int limit = (lw && rw)
      ? MAX_CORRECTION : SINGLE_WALL_MAX_CORRECTION;

  int motorLimit = min(lOff, rOff) - 5;
  if (motorLimit < 0) motorLimit = 0;
  limit = min(limit, motorLimit);

  int wanted = 0;
  if (filteredError > ERROR_DEADBAND ||
      filteredError < -ERROR_DEADBAND) {
    wanted = (int)(filteredError * CORRECTION_GAIN);
  }
  wanted = constrain(wanted, -limit, limit);

  int correction = constrain(
      wanted,
      lastCorrection - CORRECTION_STEP,
      lastCorrection + CORRECTION_STEP);
  correction = constrain(correction, -limit, limit);
  lastCorrection = correction;

  accountMotion();
  activePercent = percent;
  forwardActive = true;

  leftServo.writeMicroseconds(
      LEFT_STOP_PWM + lOff + correction);
  rightServo.writeMicroseconds(
      RIGHT_STOP_PWM - rOff + correction);
}

// ==================== 侧墙状态与长度 ====================
void updateSideTracking(const Reading &L, const Reading &R) {
  accountMotion();

  bool lo = isOpen(L);
  bool ro = isOpen(R);
  bool lw = hasWall(L);
  bool rw = hasWall(R);

  if (lw && rw) {
    enclosedBreakVotes = 0;
    enclosedNow = true;
  } else {
    enclosedNow = false;
    if (lo || ro) {
      if (enclosedBreakVotes < ENCLOSED_BREAK_COUNT) {
        enclosedBreakVotes++;
      }
      if (enclosedBreakVotes >= ENCLOSED_BREAK_COUNT) {
        enclosedForwardMs = 0;
      }
    }
  }

  if (lo) {
    if (leftOpenVotes < SIDE_OPEN_CONFIRM) leftOpenVotes++;
  } else {
    leftOpenVotes = 0;
    if (lw) leftLatched = false;
  }

  if (ro) {
    if (rightOpenVotes < SIDE_OPEN_CONFIRM) rightOpenVotes++;
  } else {
    rightOpenVotes = 0;
    if (rw) rightLatched = false;
  }
}

// ==================== 转弯 ====================
bool turnTo(int target) {
  stopRobot();
  resetCorrection();

  int relative = (target - heading + 4) % 4;
  if (relative == 0) return true;

  unsigned long duration;
  int lp, rp;

  if (relative == 1) {
    lp = SPIN_RIGHT_L;
    rp = SPIN_RIGHT_R;
    duration = RIGHT_TURN_MS;
  } else if (relative == 3) {
    lp = SPIN_LEFT_L;
    rp = SPIN_LEFT_R;
    duration = LEFT_TURN_MS;
  } else {
    lp = UTURN_LEFT_L;
    rp = UTURN_LEFT_R;
    duration = UTURN_LEFT_MS;
  }

  // 不盲目后退，避免改变转弯中心。
  leftServo.writeMicroseconds(lp);
  rightServo.writeMicroseconds(rp);

  unsigned long started = millis();
  while (millis() - started < duration) {
    if (!timeOK()) return false;
    delay(1);
  }

  stopRobot();
  delay(TURN_SETTLE_MS);
  heading = target;

  afterTurnForwardMs = 0;
  enclosedForwardMs = 0;
  enclosedNow = false;
  enclosedBreakVotes = 0;

  return timeOK();
}

// 出发后只屏蔽当前已经看见的开口，避免重复识别同一路口。
// 前墙检测不受这个屏蔽影响。
void seedSideTracking() {
  Reading L = readSensor(LEFT_TX, LEFT_RX);
  Reading R = readSensor(RIGHT_TX, RIGHT_RX);

  leftLatched = !hasWall(L);
  rightLatched = !hasWall(R);
  leftOpenVotes = rightOpenVotes = 0;
  leftCandidate = rightCandidate = false;
}

// ==================== 路口与回溯 ====================
bool depart(int direction, uint8_t kind) {
  if (!turnTo(direction)) return false;

  travelKind = kind;
  legStarted = true;
  legEquivalentMs = 0;
  legForwardMs = 0;
  phase = 0;
  resetCorrection();
  seedSideTracking();

  return true;
}

bool serviceJunction() {
  stopRobot();
  resetCorrection();

  // 关键修复：到位后重新采样，不使用接近前的结果。
  Scan s;
  bool valid = false;

  for (int retry = 0; retry < SCAN_RETRIES; retry++) {
    s = scanJunction();
    if (halted) return false;

    if (s.left >= 0 && s.front >= 0 && s.right >= 0) {
      valid = true;
      break;
    }
    delay(50);
  }

  if (!valid) {
    fail(F("JUNCTION_UNCERTAIN"));
    return false;
  }

  Serial.print(F("LFR="));
  Serial.print(s.left);
  Serial.print(',');
  Serial.print(s.front);
  Serial.print(',');
  Serial.print(s.right);
  Serial.print(F(" enclosedMs="));
  Serial.println(enclosedForwardMs);

  if (s.left == 0 && s.front == 0 && s.right == 0 &&
      enclosedForwardMs >= LONG_DEADEND_MS) {
    finishRobot();
    return false;
  }

  bool arrived = legStarted;
  int back = (heading + 2) % 4;

  if (arrived) {
    int cells = (int)((legEquivalentMs + T_CELL / 2.0f)
                      / T_CELL);
    if (cells < 1) cells = 1;

    int nx = gx + DX[heading] * cells;
    int ny = gy + DY[heading] * cells;

    if (!inside(nx, ny)) {
      fail(F("POSITION_OUT_OF_MAP"));
      return false;
    }

    gx = nx;
    gy = ny;
    legStarted = false;
  }

  Node &node = nodes[gy][gx];

  if (!node.used) {
    if (arrived && travelKind != 0) {
      fail(F("RETURN_NODE_MISMATCH"));
      return false;
    }

    node.used = 1;
    node.tried = 0;
    node.parentDir = arrived ? back : NO_PARENT;

  } else if (arrived && travelKind == 0) {
    // 新分支通到了旧节点：关闭这个环路分支，
    // 原路返回刚才出发的节点。
    node.tried |= bitFor(back);
    return depart(back, 2);
  }

  bool open[4] = {false, false, false, false};
  open[(heading + 3) % 4] = s.left == 1;
  open[heading] = s.front == 1;
  open[(heading + 1) % 4] = s.right == 1;

  if (arrived) {
    open[back] = true;   // 实际来路
    node.tried |= bitFor(back);
  }

  int order[3] = {
    (heading + 3) % 4,
    heading,
    (heading + 1) % 4
  };

  for (int i = 0; i < 3; i++) {
    int d = order[i];
    if (open[d] &&
        !(node.tried & bitFor(d)) &&
        d != node.parentDir) {
      node.tried |= bitFor(d);
      return depart(d, 0);
    }
  }

  if (node.parentDir != NO_PARENT) {
    int d = node.parentDir;

    if (!open[d]) {
      fail(F("PARENT_DIRECTION_BLOCKED"));
      return false;
    }

    return depart(d, 1);
  }

  // 搜完仍没识别出终点，不伪报FINISH。
  fail(F("NO_UNTRIED_ROUTE_OR_FINISH"));
  return false;
}

// ==================== 调试输出 ====================
void printStatus(const Reading &L, const Reading &F,
                 const Reading &R) {
  if (millis() - lastPrintAt < 300) return;
  lastPrintAt = millis();

  Serial.print(F("P="));
  Serial.print(phase);
  Serial.print(F(" L="));
  Serial.print(L.bands);
  Serial.print('/');
  Serial.print(L.votes);
  Serial.print(F(" F="));
  Serial.print(F.bands);
  Serial.print('/');
  Serial.print(F.votes);
  Serial.print(F(" R="));
  Serial.print(R.bands);
  Serial.print('/');
  Serial.print(R.votes);
  Serial.print(F(" wallMs="));
  Serial.println(enclosedForwardMs);
}

// ==================== 初始化 ====================
void setup() {
  Serial.begin(115200);

  pinMode(LEFT_TX, OUTPUT);
  pinMode(FRONT_TX, OUTPUT);
  pinMode(RIGHT_TX, OUTPUT);
  digitalWrite(LEFT_TX, LOW);
  digitalWrite(FRONT_TX, LOW);
  digitalWrite(RIGHT_TX, LOW);

  pinMode(LEFT_RX, INPUT);
  pinMode(FRONT_RX, INPUT);
  pinMode(RIGHT_RX, INPUT);

  pinMode(LED_L, OUTPUT);
  pinMode(LED_M, OUTPUT);
  pinMode(LED_R, OUTPUT);

  leftServo.attach(LEFT_SERVO);
  rightServo.attach(RIGHT_SERVO);

  accountAt = millis();
  stopRobot();
  leds(false, false, false);
  delay(1000);

  memset(nodes, 0, sizeof(nodes));

  for (int y = 0; y < GRID; y++) {
    for (int x = 0; x < GRID; x++) {
      nodes[y][x].parentDir = NO_PARENT;
    }
  }

  runStartedAt = millis();
  accountAt = millis();

  // 起点按已知摆放处理：面朝通道，先向前探索。
  // 起点后方不作为未探索分支。
  Node &root = nodes[gy][gx];
  root.used = 1;
  root.parentDir = NO_PARENT;
  root.tried = bitFor(heading);

  legStarted = true;
  travelKind = 0;
  afterTurnForwardMs = SIDE_REARM_MS;
  seedSideTracking();

  Serial.println(F("READY: continuous corridor exploration"));
}

// ==================== 主循环 ====================
void loop() {
  if (halted) {
    stopRobot();

    if (finished) {
      leds(true, true, true);
    } else {
      bool on = (millis() / 250UL) % 2UL;
      leds(on, on, on);
    }

    delay(20);
    return;
  }

  if (!timeOK()) return;

  Reading F = readSensor(FRONT_TX, FRONT_RX);

  // 过近时立即停轮，再读取左右，避免继续运动。
  bool close = tooClose(F);
  if (close) stopRobot();

  Reading L = readSensor(LEFT_TX, LEFT_RX);
  Reading R = readSensor(RIGHT_TX, RIGHT_RX);

  updateSideTracking(L, R);

  if (close) {
    serviceJunction();
    return;
  }

  bool leftEvent =
      leftOpenVotes >= SIDE_OPEN_CONFIRM && !leftLatched;
  bool rightEvent =
      rightOpenVotes >= SIDE_OPEN_CONFIRM && !rightLatched;

  // 已进入侧开口居中阶段：
  // 保持纠偏补走，不因普通前墙信号立刻转。
  if (phase == 2) {
    if (millis() - phaseStartedAt >= SIDE_CENTER_MS) {
      stopRobot();

      // 若原先识别的开口在补走后已消失，不能盲转。
      bool candidateStillOpen =
          (leftCandidate && isOpen(L)) ||
          (rightCandidate && isOpen(R));

      if (!candidateStillOpen) {
        fail(F("OPENING_LOST_CHECK_SIDE_CENTER_MS"));
        return;
      }

      serviceJunction();
      return;
    }

    driveCorrected(L, R, FRONT_APPROACH_PCT);
    printStatus(L, F, R);
    return;
  }

  // 正常或前墙接近过程中发现新侧开口：
  // 从开口事件开始做车轴居中补走，
  // 不把完整前墙补偿时间再叠加一遍。
  if ((leftEvent || rightEvent) &&
      afterTurnForwardMs >= SIDE_REARM_MS) {
    leftCandidate = leftEvent;
    rightCandidate = rightEvent;

    if (leftEvent) leftLatched = true;
    if (rightEvent) rightLatched = true;

    phase = 2;
    phaseStartedAt = millis();

    driveCorrected(L, R, FRONT_APPROACH_PCT);
    return;
  }

  // 前墙接近阶段：继续带纠偏行驶。
  if (phase == 1) {
    if (millis() - phaseStartedAt >= FRONT_APPROACH_MS) {
      serviceJunction();
      return;
    }

    driveCorrected(L, R, FRONT_APPROACH_PCT);
    printStatus(L, F, R);
    return;
  }

  // 普通报墙只启动接近阶段，绝不直接转。
  if (F.bands >= FRONT_WALL_THRESHOLD) {
    phase = 1;
    phaseStartedAt = millis();

    driveCorrected(L, R, FRONT_APPROACH_PCT);
    return;
  }

  // 正常通道：连续行驶，不每格停车。
  bool slow =
      !hasWall(L) || !hasWall(R) ||
      F.bands >= 1 || F.votes >= 3 ||
      filteredError > 3.0f || filteredError < -3.0f;

  driveCorrected(L, R,
                 slow ? CAUTION_DRIVE_PERCENT : 100);

  leds(false, true, false);
  printStatus(L, F, R);
}
```

**你最关心的两个动作，这版是这样执行的：**

| 情况 | 动作 |
|---|---|
| 前墙提前一格报墙，侧面还没缺口 | 连续带纠偏前进，不立即转 |
| 右缺口到达、左边仍有墙 | 以左墙纠偏，补走 `SIDE_CENTER_MS`，重新采样后再决定 |
| 左缺口到达、右边仍有墙 | 对称处理 |
| 补走中前墙已过近 | 停轮重新判断，不能为了补足时间继续顶墙 |
| U 形弯第一弯结束 | 继续前进；前墙再报墙仍走接近流程，不直接执行第二转 |
| 长死路末端 | 三向有墙且双墙直段时间够长，停车三灯常亮 |

**不要改你的已复测转弯时间。先重点核对 `SIDE_CENTER_MS = 180` 和 `LONG_DEADEND_MS = 2*T_CELL`：这两个是待实测初值，不能冒充已经准确。**

另外，这版仍不是经过编译和实车验证的成品保证。**连续行驶的时间定位在往返速度不一致时可能把同一岔口记到不同坐标，这是目前回溯可靠性的实际限制；我不能用“逻辑检查过了”掩盖这一点。**
